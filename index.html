/** * ===== ตั้งค่าก่อนใช้งาน ===== * 1. เปิด Google Sheet ที่จะใช้เก็บข้อมูล > คัดลอก Sheet ID จาก URL * ตัวอย่าง URL: https://docs.google.com/spreadsheets/d/[SHEET_ID_ตรงนี้]/edit * 2. เปิด/สร้างโฟลเดอร์ Google Drive สำหรับเก็บรูปใบเสร็จ > คัดลอก Folder ID จาก URL * ตัวอย่าง URL: https://drive.google.com/drive/folders/[FOLDER_ID_ตรงนี้] * 3. วาง ID ทั้งสองด้านล่างนี้ (แทนที่ข้อความ 'ใส่ ... ตรงนี้') * 4. const FOLDER_ID = '1zPE3fe7M0fZz4-q7-nZfaf1FmqAd0Z3i'; const SHEET_NAME = 'รายการเงินโครงการ'; const CATEGORY_LABELS = { 'cot': 'COT จ่ายเอง', 'fund-in': 'เงินเข้ากองกลาง', 'fund-out': 'รายจ่ายกองกลาง' }; const REVERSE_CATEGORY_LABELS = {}; Object.keys(CATEGORY_LABELS).forEach(function (k) { REVERSE_CATEGORY_LABELS[CATEGORY_LABELS[k]] = k; }); const PAYER_LABELS = { a: 'COT', b: 'TMCC', c: 'CPP' }; const REVERSE_PAYER_LABELS = {}; Object.keys(PAYER_LABELS).forEach(function (k) { REVERSE_PAYER_LABELS[PAYER_LABELS[k]] = k; }); // column order used everywhere below const HEADERS = [ 'ID', 'วันที่บันทึก', 'ประเภท', 'รายการ', 'จำนวนเงิน', 'สถานที่', 'COT (47%)', 'TMCC (32%)', 'CPP (21%)', 'ลิงก์ใบเสร็จ/สลิป', 'สถานะ', 'วันที่ลบ', 'ผู้จ่ายจริง (ออกใบกำกับภาษี)' ]; const STATUS_COL = HEADERS.indexOf('สถานะ') + 1; // 1-based column index const DELETED_AT_COL = HEADERS.indexOf('วันที่ลบ') + 1; // 1-based column index const DELETED_LABEL = 'ลบแล้ว'; function getSheet_() { const ss = SpreadsheetApp.openById(SHEET_ID); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow(HEADERS); sheet.setFrozenRows(1); } else { ensureHeaders_(sheet); // เผื่อชีทเดิมยังไม่มีคอลัมน์ สถานะ/วันที่ลบ ให้เติมให้อัตโนมัติ } return sheet; } // เติม header คอลัมน์ที่ยังขาดต่อท้าย โดยไม่กระทบคอลัมน์/ข้อมูลเดิมที่มีอยู่แล้ว function ensureHeaders_(sheet) { const lastCol = sheet.getLastColumn(); if (lastCol < HEADERS.length) { sheet.getRange(1, lastCol + 1, 1, HEADERS.length - lastCol) .setValues([HEADERS.slice(lastCol)]); } } /** * ===== รันเองครั้งเดียว ===== * ใช้เพิ่มแถวหัวตาราง (ID, วันที่บันทึก, ประเภท, ...) ให้ Sheet ที่ยังไม่มีหัวตาราง * วิธีรัน: ใน Apps Script Editor เลือกฟังก์ชัน "addHeaderRow" จาก dropdown ข้างปุ่ม Run * แล้วกด Run (ครั้งแรกจะขอ authorize สิทธิ์เข้าถึง Sheet ให้กด Allow) * ถ้า Sheet มีหัวตารางที่ถูกต้องอยู่แล้ว ฟังก์ชันนี้จะไม่ทำอะไรซ้ำ */ function addHeaderRow() { const sheet = getSheet_(); const firstCell = sheet.getRange(1, 1).getValue(); if (firstCell === HEADERS[0]) { Logger.log('มีหัวตารางอยู่แล้ว ไม่ต้องทำอะไรเพิ่ม'); return; } sheet.insertRowBefore(1); sheet.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]); sheet.getRange(1, 1, 1, HEADERS.length).setFontWeight('bold').setBackground('#f4f0ff'); sheet.setFrozenRows(1); Logger.log('เพิ่มหัวตารางเรียบร้อยแล้ว'); } function doPost(e) { try { const data = JSON.parse(e.postData.contents); const sheet = getSheet_(); if (data.action === 'delete') { return softDeleteRow_(sheet, data.id); } if (data.action === 'clearAll') { return softClearAll_(sheet); } let receiptUrl = ''; if (data.imageBase64) { const folder = DriveApp.getFolderById(FOLDER_ID); const matches = data.imageBase64.match(/^data:(.*);base64,(.*)$/); if (matches) { const contentType = matches[1]; const base64Data = matches[2]; const safeName = (data.itemName || 'receipt').replace(/[^\u0E00-\u0E7Fa-zA-Z0-9]/g, '_'); const fileName = safeName + '_' + Utilities.formatDate(new Date(), 'GMT+7', 'yyyyMMdd_HHmmss') + '.jpg'; const blob = Utilities.newBlob(Utilities.base64Decode(base64Data), contentType, fileName); const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); receiptUrl = file.getUrl(); } } const categoryLabel = CATEGORY_LABELS[data.category] || data.category || ''; const breakdown = data.breakdown || {}; const id = data.id || Utilities.getUuid(); sheet.appendRow([ id, new Date(), categoryLabel, data.itemName || '', data.amount || 0, data.site || '', breakdown.a !== undefined ? breakdown.a : '', breakdown.b !== undefined ? breakdown.b : '', breakdown.c !== undefined ? breakdown.c : '', receiptUrl, '', // สถานะ - ว่างไว้ตอนสร้างใหม่ '', // วันที่ลบ - ว่างไว้ตอนสร้างใหม่ PAYER_LABELS[data.payer] || '' // ผู้จ่ายจริง (เฉพาะ fund-out) ]); return ContentService.createTextOutput(JSON.stringify({ success: true, id: id, receiptUrl: receiptUrl })).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.toString() })).setMimeType(ContentService.MimeType.JSON); } } // mark แถวเดียวเป็น "ลบแล้ว" แทนการลบแถวจริง function softDeleteRow_(sheet, id) { try { const values = sheet.getDataRange().getValues(); for (let i = 1; i < values.length; i++) { if (String(values[i][0]) === String(id)) { sheet.getRange(i + 1, STATUS_COL).setValue(DELETED_LABEL); sheet.getRange(i + 1, DELETED_AT_COL).setValue(new Date()); return ContentService.createTextOutput(JSON.stringify({ success: true })) .setMimeType(ContentService.MimeType.JSON); } } return ContentService.createTextOutput(JSON.stringify({ success: false, error: 'ไม่พบรายการที่จะลบ' })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } // mark ทุกแถวที่ยังไม่ถูกลบ ให้เป็น "ลบแล้ว" ทั้งหมด แทนการลบแถวจริง function softClearAll_(sheet) { try { const lastRow = sheet.getLastRow(); if (lastRow > 1) { const rowCount = lastRow - 1; const statusValues = sheet.getRange(2, STATUS_COL, rowCount, 1).getValues(); const now = new Date(); const newStatus = []; const newDeletedAt = []; for (let i = 0; i < rowCount; i++) { if (statusValues[i][0] === DELETED_LABEL) { // ที่ลบไปแล้วก่อนหน้านี้ ปล่อยตามเดิม ไม่ต้องแตะวันที่ลบซ้ำ newStatus.push([DELETED_LABEL]); newDeletedAt.push([sheet.getRange(2 + i, DELETED_AT_COL).getValue()]); } else { newStatus.push([DELETED_LABEL]); newDeletedAt.push([now]); } } sheet.getRange(2, STATUS_COL, rowCount, 1).setValues(newStatus); sheet.getRange(2, DELETED_AT_COL, rowCount, 1).setValues(newDeletedAt); } return ContentService.createTextOutput(JSON.stringify({ success: true })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { try { const sheet = getSheet_(); const values = sheet.getDataRange().getValues(); const rows = values.slice(1) .filter(function (r) { return r[0]; }) // skip any blank rows .filter(function (r) { return r[STATUS_COL - 1] !== DELETED_LABEL; }) // skip soft-deleted rows .map(function (r) { return { id: r[0], date: r[1], category: REVERSE_CATEGORY_LABELS[r[2]] || r[2], itemName: r[3], amount: r[4], site: r[5], a: r[6], b: r[7], c: r[8], receiptUrl: r[9], payer: REVERSE_PAYER_LABELS[r[12]] || '' }; }); return ContentService.createTextOutput(JSON.stringify({ success: true, items: rows })).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.toString() })).setMimeType(ContentService.MimeType.JSON); } }
